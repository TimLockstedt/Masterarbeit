else CASE == 2
    fname = 'complex';
    im = double(rgb2gray(imread('peppers.png')));
    im=imresize(im,[ss,ss]);
    im = im/max(im,[],"all");
    I1(I1 == 0) = 0.05;
    t = I1;
    theta_map = 1*p_im;
    p = I2* 2 * pi;
    delta_map = 1*im * delta;
end
